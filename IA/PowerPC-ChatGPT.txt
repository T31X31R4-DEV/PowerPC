bool openFS(void)

bool writeFile(String state, String path)
{
  File rFile = SPIFFS.open(path, "w+");

  if(!rFile)
    return false;

  rFile.println(state);

  rFile.close();

  return true;
}

bool appendFile(String state, String path)
{
  File rFile = SPIFFS.open(path, "a+");

  if(!rFile)
    return false;

  rFile.println(state);

  rFile.close();

  return true;
}

String readFile(String path)
{
  File rFile = SPIFFS.open(path, "r");

  if(!rFile)
    return "";

  String content = rFile.readString();

  rFile.close();

  return content;
}

bool gravaConfigEx(String a, String b, String c, String d, String e, String f, String g, String h, String i)
{
  bool ok = true;

  ok &= writeFile(a, ARQ_TEMP_SENSOR);
  ok &= appendFile(b, ARQ_TEMP_SENSOR);
  ok &= appendFile(c, ARQ_TEMP_SENSOR);
  ok &= appendFile(d, ARQ_TEMP_SENSOR);
  ok &= appendFile(e, ARQ_TEMP_SENSOR);
  ok &= appendFile(f, ARQ_TEMP_SENSOR);
  ok &= appendFile(g, ARQ_TEMP_SENSOR);
  ok &= appendFile(h, ARQ_TEMP_SENSOR);
  ok &= appendFile(i, ARQ_TEMP_SENSOR);

  if (ok)
  {
    SPIFFS.remove(ARQ_SENSOR);
    SPIFFS.rename(ARQ_TEMP_SENSOR, ARQ_SENSOR);
  }

  return ok;
}

void carregaConfigEx()
{
  if(!SPIFFS.exists(ARQ_SENSOR))
    return;

  File rFile = SPIFFS.open(ARQ_SENSOR, "r");

  if (!rFile)
    return;

  dev.Host              = rFile.readStringUntil('\n');
  dev.IPMqtt            = rFile.readStringUntil('\n');
  dev.Porta             = rFile.readStringUntil('\n');
  dev.Mqttuser          = rFile.readStringUntil('\n');
  dev.Mqttpass          = rFile.readStringUntil('\n');
  dev.MqttNomeAmigavel  = rFile.readStringUntil('\n');
  dev.MqttTopico        = rFile.readStringUntil('\n');
  dev.WebHookUrl        = rFile.readStringUntil('\n');
  dev.TempoParaDesligar = rFile.readStringUntil('\n');

  dev.Host.trim();
  dev.IPMqtt.trim();
  dev.Porta.trim();
  dev.Mqttuser.trim();
  dev.Mqttpass.trim();
  dev.MqttNomeAmigavel.trim();
  dev.MqttTopico.trim();
  dev.WebHookUrl.trim();
  dev.TempoParaDesligar.trim();

  rFile.close();
}

void setupWiFiAuto()
{
  const char *ssid = SSID_NOME;  // Para o AP para a configuração inicial
  char ssidX[40];                // SSID de configuração (PowerPC-XXXXXX)
  char Host[33] = SSID_NOME;
  char IPMqtt[256] = "127.0.0.1";
  char Porta[7] = "1883";
  char Mqttuser[65] = "mqtt";
  char Mqttpass[65] = "mqtt";
  char MqttNomeAmigavel[65] = TEXTO_ABERTURA;
  char MqttTopico[65] = TEXTO_ABERTURA;
  char WebHookUrl[256]= WEBHOOK_URL;
  char TempoParaDesligar[2] = "1";

  carregaConfigEx();

  if (dev.Host.length())
    dev.Host.toCharArray(Host, 33);

  if (dev.IPMqtt.length())
    dev.IPMqtt.toCharArray(IPMqtt, 256);

  if (dev.Porta.length())
    dev.Porta.toCharArray(Porta, 7);

  if (dev.Mqttuser.length())
    dev.Mqttuser.toCharArray(Mqttuser, 65);

  if (dev.Mqttpass.length())
    dev.Mqttpass.toCharArray(Mqttpass, 65);

  if (dev.MqttNomeAmigavel.length())
    dev.MqttNomeAmigavel.toCharArray(MqttNomeAmigavel, 65);

  if (dev.MqttTopico.length())
    dev.MqttTopico.toCharArray(MqttTopico, 65);

  if (dev.WebHookUrl.length())
    dev.WebHookUrl.toCharArray(WebHookUrl, sizeof(WebHookUrl));

  if (dev.TempoParaDesligar.length())
    dev.TempoParaDesligar.toCharArray(TempoParaDesligar, 2);

  WiFiManagerParameter p1("host", "Host", Host, 32);
  WiFiManagerParameter p2("IPMqtt", "IPMqtt", IPMqtt, 255);
  WiFiManagerParameter p3("Porta", "Porta", Porta, 6);
  WiFiManagerParameter p4("Mqttuser", "Mqttuser", Mqttuser, 64);
  WiFiManagerParameter p5("Mqttpass", "Mqttpass", Mqttpass, 64);
  WiFiManagerParameter p6("MqttNomeAmigavel", "MqttNomeAmigavel", MqttNomeAmigavel, 64);
  WiFiManagerParameter p7("MqttTopico", "MqttTopico", MqttTopico, 64);
  WiFiManagerParameter p8("WebHookUrl", "WebHookUrl", WebHookUrl, 255);
  WiFiManagerParameter p9("TempoParaDesligar", "TempoParaDesligar", TempoParaDesligar, 1);

  wfm.addParameter(&p1);
  wfm.addParameter(&p2);
  wfm.addParameter(&p3);
  wfm.addParameter(&p4);
  wfm.addParameter(&p5);
  wfm.addParameter(&p6);
  wfm.addParameter(&p7);
  wfm.addParameter(&p8);
  wfm.addParameter(&p9);

  wfm.setSaveConfigCallback(saveConfigCallback);

  //String clientId = "esp8266-" + String(ESP.getChipId());
  //clientId.c_str(),
    
  String mac = WiFi.macAddress();
  mac.replace(":", "");
  String apName = String(ssid) + "-" + mac.substring(mac.length() > 6 ? mac.length() - 6 : 0);
  apName.toCharArray(ssidX, sizeof(ssidX));

  if (!wfm.autoConnect(ssidX, PASS_NOME))
    ESP.restart();

  if (shouldSaveConfig)
  {
    gravaConfigEx(p1.getValue(), p2.getValue(), p3.getValue(), p4.getValue(), p5.getValue(), p6.getValue(), p7.getValue(), p8.getValue(), p9.getValue());
    ESP.restart();
  }
}

// =====================================================
// MQTT / COMPATIBILIDADE COM TASMOTA + HOME ASSISTANT
// Mantém os tópicos antigos esp8266/<topico>/... e acrescenta:
// cmnd/<topico>/POWER, stat/<topico>/POWER, tele/<topico>/STATE e LWT.
// =====================================================

String topicoBase()
{
  String base = dev.MqttTopico;
  base.trim();
  if (base.length() == 0)
    base = TEXTO_ABERTURA;
  base.replace(" ", "_");
  return base;
}

String topicoLegado()
{
  return "esp8266/" + topicoBase();
}

String topicoComandoTasmota()
{
  return "cmnd/" + topicoBase() + "/#";
}

String topicoEstadoTasmota()
{
  return "stat/" + topicoBase() + "/POWER";
}

String topicoTelemetriaTasmota()
{
  return "tele/" + topicoBase() + "/STATE";
}

String topicoLWT()
{
  return "tele/" + topicoBase() + "/LWT";
}

String jsonEscape(const String &valor)
{
  String saida;
  saida.reserve(valor.length() + 8);
  for (unsigned int i = 0; i < valor.length(); i++)
  {
    char c = valor[i];
    if (c == '"' || c == '\\')
      saida += '\\';
    if (c == '\n')
      saida += "\\n";
    else if (c == '\r')
      saida += "\\r";
    else
      saida += c;
  }
  return saida;
}

void PublicaEstadoMQTT()
{
  if (!ConexaoMQTT.connected())
    return;

  const char *estado = releLigado ? "ON" : "OFF";
  String legado = topicoLegado() + "/state";
  String tasmota = topicoEstadoTasmota();

  ConexaoMQTT.publish(legado.c_str(), estado, true);
  ConexaoMQTT.publish(tasmota.c_str(), estado, true);

  String power = "{\"POWER\":\"" + String(estado) + "\"}";
  String tele = topicoTelemetriaTasmota();
  ConexaoMQTT.publish(tele.c_str(), power.c_str(), true);
}

void publicaTelemetriaMQTT()
{
  if (!ConexaoMQTT.connected())
    return;

  String payload = "{\"Time\":\"";
  time_t now = time(nullptr);
  payload += String((unsigned long)now);
  payload += "\",\"Uptime\":";
  payload += String(millis() / 1000UL);
  payload += ",\"POWER\":\"";
  payload += (releLigado ? "ON" : "OFF");
  payload += "\",\"Wifi\":{\"AP\":1,\"SSId\":\"";
  payload += jsonEscape(WiFi.SSID());
  payload += "\",\"RSSI\":";
  payload += String(WiFi.RSSI());
  payload += "},\"Hostname\":\"";
  payload += jsonEscape(dev.Host);
  payload += "\"}";

  String topic = topicoTelemetriaTasmota();
  ConexaoMQTT.publish(topic.c_str(), payload.c_str(), true);
}

void publicaDiscoveryHomeAssistant()
{
  if (!ConexaoMQTT.connected())
    return;

  String id = String(ESP.getChipId(), HEX);
  id.toLowerCase();

  String discoveryTopic = "homeassistant/switch/" + topicoLegado() + "/config";
  String commandTopic = "cmnd/" + topicoBase() + "/POWER";
  String stateTopic = topicoEstadoTasmota();
  String availabilityTopic = topicoLWT();

  String nome = dev.MqttNomeAmigavel;
  if (nome.length() == 0)
    nome = TEXTO_ABERTURA;

  String payload = "{";
  payload += "\"name\":\"" + jsonEscape(nome) + "\",";
  payload += "\"unique_id\":\"powerpc_" + id + "_power\",";
  payload += "\"icon\":\"mdi:power\",";
  payload += "\"command_topic\":\"" + commandTopic + "\",";
  payload += "\"state_topic\":\"" + stateTopic + "\",";
  payload += "\"payload_on\":\"ON\",\"payload_off\":\"OFF\",";
  payload += "\"state_on\":\"ON\",\"state_off\":\"OFF\",";
  payload += "\"availability_topic\":\"" + availabilityTopic + "\",";
  payload += "\"payload_available\":\"Online\",\"payload_not_available\":\"Offline\",";
  payload += "\"optimistic\":false,";
  payload += "\"device\":{";
  payload += "\"identifiers\":[\"powerpc_" + id + "\"],";
  payload += "\"name\":\"" + jsonEscape(nome) + "\",";
  payload += "\"manufacturer\":\"PowerPC / ESP8266\",";
  payload += "\"model\":\"PowerPC ESP8266\",";
  payload += "\"sw_version\":\"" + String(VERSAO) + "\"";
  payload += "}}";

  // Retained: Home Assistant redescobre o dispositivo após reinicializações.
  ConexaoMQTT.publish(discoveryTopic.c_str(), payload.c_str(), true);
}

void setRele(bool ligar)
{
  releLigado = ligar;
  digitalWrite(PIN_RELE, ligar ? ON : OFF);
  PublicaEstadoMQTT();
}

void executaWebhook()
{
  if (WiFi.status() != WL_CONNECTED || dev.WebHookUrl.length() == 0)
    return;

  WiFiClient client;
  HTTPClient http;
  http.setTimeout(3000);

  if (http.begin(client, dev.WebHookUrl))
  {
    int httpCode = http.GET();
    // Evita manter a conexão aberta; a funcionalidade do webhook é preservada.
    (void)httpCode;
    http.end();
  }
}

void solicitaLigamento()
{
  desligamentoPendente = false;
  setRele(true);
}

void solicitaDesligamento()
{
  long minutosConfigurados = dev.TempoParaDesligar.toInt();
  if (minutosConfigurados < 0) minutosConfigurados = 0;
  unsigned long tempo = (unsigned long)minutosConfigurados * MINUTO;

  if (!releLigado)
    return;

  desligamentoPendente = true;
  horarioDesligamento = millis() + tempo;

  executaWebhook();

  // Compatibilidade com clientes legados: anuncia OFF enquanto aguarda o timer,
  // como fazia o firmware original.
  String legado = topicoLegado() + "/state";
  if (ConexaoMQTT.connected())
    ConexaoMQTT.publish(legado.c_str(), "OFF", true);
}

void recebendoMensagensMQTT(char* topic, byte* payload, unsigned int length)
{
  String mensagem;
  mensagem.reserve(length + 1);

  for (unsigned int i = 0; i < length; i++)
    mensagem += (char)payload[i];

  mensagem.trim();
  mensagem.toUpperCase();

  String recebido(topic);
  String base = topicoBase();
  String cmdPrefix = "cmnd/" + base + "/";

  // Mantém o tópico legado /set e também aceita o padrão cmnd/<device>/POWER.
  if (recebido == topicoLegado() + "/set" ||
      recebido == cmdPrefix + "POWER" ||
      recebido == cmdPrefix + "POWER1")
  {
    if (mensagem == "ON")
      mqttLiga = true;
    else if (mensagem == "OFF")
      mqttDesliga = true;
    else if (mensagem == "TOGGLE")
    {
      if (releLigado || desligamentoPendente)
        mqttDesliga = true;
      else
        mqttLiga = true;
    }
    return;
  }

  if (recebido == cmdPrefix + "STATUS" || recebido == cmdPrefix + "STATE")
  {
    PublicaEstadoMQTT();
    publicaTelemetriaMQTT();
    return;
  }

  if (recebido == cmdPrefix + "RESTART" && (mensagem == "1" || mensagem == "RESTART"))
  {
    ESP.restart();
  }
}

bool reconectaMQTT()
{
  if (ConexaoMQTT.connected())
    return true;

  if (WiFi.status() != WL_CONNECTED)
    return false;

  unsigned long agora = millis();
  if (agora - ultimaTentativaMQTT < INTERVALO_RECONEXAO_MQTT)
    return false;
  ultimaTentativaMQTT = agora;

  ConexaoMQTT.setServer(dev.IPMqtt.c_str(), dev.Porta.toInt());
  ConexaoMQTT.setCallback(recebendoMensagensMQTT);
  ConexaoMQTT.setBufferSize(768);
  ConexaoMQTT.setKeepAlive(30);
  ConexaoMQTT.setSocketTimeout(3);

  String clientId = "PowerPC-" + String(ESP.getChipId(), HEX);
  bool conectou;
  if (dev.Mqttuser.length() > 0)
    conectou = ConexaoMQTT.connect(clientId.c_str(),
                                   dev.Mqttuser.c_str(),
                                   dev.Mqttpass.c_str(),
                                   topicoLWT().c_str(), 0, true, "Offline");
  else
    conectou = ConexaoMQTT.connect(clientId.c_str(),
                                   topicoLWT().c_str(), 0, true, "Offline");

  if (conectou)
  {
    String topicoComandoLegado = topicoLegado() + "/set";
    ConexaoMQTT.subscribe(topicoComandoLegado.c_str());

    String topicoComando = topicoComandoTasmota();
    ConexaoMQTT.subscribe(topicoComando.c_str());

    String lwt = topicoLWT();
    ConexaoMQTT.publish(lwt.c_str(), "Online", true);

    publicaDiscoveryHomeAssistant();
    PublicaEstadoMQTT();
    publicaTelemetriaMQTT();

    ultimoDiscoveryMQTT = agora;
    ultimaTelemetriaMQTT = agora;
  }

  return conectou;
}

void handleRoot()
{
  char tmp[192];
  time_t now = time(nullptr);
  struct tm t;
  localtime_r(&now, &t);  // mais seguro e levemente mais eficiente

  String html = "<html><meta http-equiv='refresh' content='5'><body><pre>";

  html += TEXTO_ABERTURA;
  #ifdef ESP01
      html += " - ESP-01/";
  #else
      html += " - ESP-8266/";
  #endif
  
  html +=  ARDUINO_BOARD;

  html += " v";
  html += VERSAO;
  html += "\n";
  html += "\n";

  html += "IP: ";
  html += WiFi.localIP().toString();
  html += "\n";

  html += "WebHook: ";
  html += dev.WebHookUrl;
  html += "\n";

  html += "Espera: ";
  html += dev.TempoParaDesligar;
  html += "\n";

  snprintf(tmp, sizeof(tmp),
    "Data/Hora: %02d/%02d/%04d %02d:%02d:%02d",
    t.tm_mday,
    t.tm_mon + 1,
    t.tm_year + 1900,
    t.tm_hour,
    t.tm_min,
    t.tm_sec
  );
  html += tmp;

  html += "\n";
  html += "\n";

  html += "Estado: ";

  if (desligamentoPendente)
    html += "DESLIGAMENTO_AGENDADO";
  else
    html += releLigado ? "ON" : "OFF";

  html += "</pre></body></html>";

  server.send(200, "text/html", html);
}

void handleStatus()
{
  if (server.hasArg("rele"))
  {
    String valor = server.arg("rele");

    valor.toUpperCase();

    if (valor == "ON")
      solicitaLigamento();
    else if (valor == "OFF")
    { 
      if (server.hasArg("timer"))
      {
        String tempo = server.arg("timer");
        long minutos = tempo.toInt();
        if (minutos < 0) minutos = 0;
        if (releLigado)
        {
          desligamentoPendente = true;
          horarioDesligamento = millis() + ((unsigned long)minutos * MINUTO);
          executaWebhook();
        }
      }
      else
      {
        desligamentoPendente = false;
        setRele(false); // desligamneto imediato
      }
    }
  }

  if (desligamentoPendente)
    server.send(200, "text/plain", "DESLIGAMENTO_AGENDADO");
  else
    server.send(200, "text/plain", releLigado ? "ON" : "OFF");
}

void handleStatusJson() {
  char dataHora[32];
  String estado;

  time_t now = time(nullptr);
  struct tm t;
  localtime_r(&now, &t);

  if (desligamentoPendente)
    estado = "DESLIGAMENTO_AGENDADO";
  else
    estado = releLigado ? "ON" : "OFF";

  snprintf(dataHora, sizeof(dataHora),
    "%02d/%02d/%04d %02d:%02d:%02d",
    t.tm_mday, t.tm_mon + 1, t.tm_year + 1900,
    t.tm_hour, t.tm_min, t.tm_sec);

  String json = "{\"estado\":\"" + estado + "\",\"DataHora\":\"" + String(dataHora) +
                "\",\"rele\":\"" + String(releLigado ? "ON" : "OFF") +
                "\",\"desligamento_pendente\":" + String(desligamentoPendente ? "true" : "false") +
                ",\"ip\":\"" + WiFi.localIP().toString() +
                "\",\"rssi\":" + String(WiFi.RSSI()) +
                ",\"mqtt\":" + String(ConexaoMQTT.connected() ? "true" : "false") + "}";
  server.send(200, "application/json", json);
}

void verificaBotaoReset()
{
#ifdef ESP01
// se for ESP01... não faz nada, não tem Reset fisico
#else
  if (digitalRead(PIN_BOTAO_RESET) == LOW)
  {
    if (tempoBotaoReset == 0)
      tempoBotaoReset = millis();
    
    if (millis() - tempoBotaoReset > 10000)
      handleFactoryReset();
  }
  else
    tempoBotaoReset = 0;
#endif
}

void lerBotao()
{
  unsigned long tempoAtual = millis();

  if (tempoAtual - ultimaLeituraBotao < 100)
    return;

  ultimaLeituraBotao = tempoAtual;

  int leitura = digitalRead(PIN_BOTAO);

  if (leitura != UltimoValor)
  {
    UltimoValor = leitura;

    if (desligamentoPendente)
      solicitaLigamento();
    else if (releLigado)
      solicitaDesligamento();
    else
      solicitaLigamento();

    ultimoEnvioAutomatico = tempoAtual;
  }
}

void WatchdogMQTT()
{
  static unsigned long ultimaPublicacao = 0;
  unsigned long agora = millis();

  if (agora - ultimaPublicacao < 30000UL)
    return;

  ultimaPublicacao = agora;
  if (!desligamentoPendente)
    PublicaEstadoMQTT();
}

void sendMessage(const char *MESSAGE)
{
  String MSG = MESSAGE;
  time_t now = time(nullptr);

  String payload = "{";
         payload += "\"msg\":\"" + MSG + "\",";
         payload += "\"source\":\"esp8266\",";
         payload += "\"device\":\"sensor_sala\",";
         payload += "\"ip\":\"" + WiFi.localIP().toString() + "\",";
         payload += "\"ts\":\"" + String(now) + "\"";
         payload += "}";

  ConexaoMQTT.publish("instagram/send", payload.c_str());
}

// =====================================================
// SETUP
// =====================================================

void setup()
{
  Serial.begin(BAUD_RATE);
  Serial.end();
  
  openFS();

  //wfm.resetSettings();  //
  //SPIFFS.format();      //
  //ESP.restart();        //

  setupWiFiAuto();

  server.on("/",                     handleRoot);
  server.on("/status",               handleStatus);
  server.on("/reboot",               handleReboot);
  server.on("/factoryreset",         handleFactoryReset);
  server.on("/status/json",          handleStatusJson);
  server.on("/cm", []() {
    String cmnd = server.arg("cmnd");
    cmnd.trim();
    cmnd.toUpperCase();
    if (cmnd == "POWER ON" || cmnd == "POWER1 ON" || cmnd == "ON")
      solicitaLigamento();
    else if (cmnd == "POWER OFF" || cmnd == "POWER1 OFF" || cmnd == "OFF")
      solicitaDesligamento();
    else if (cmnd == "POWER TOGGLE" || cmnd == "TOGGLE")
    {
      if (releLigado || desligamentoPendente) solicitaDesligamento();
      else solicitaLigamento();
    }
    else if (cmnd == "STATUS" || cmnd == "STATUS 0")
    {
      handleStatusJson();
      return;
    }
    else
    {
      server.send(400, "text/plain", "Comando nao reconhecido");
      return;
    }
    handleStatus();
  });
  server.on("/health", []() {
    String body = "{\"ready\":true,\"device\":\"" + jsonEscape(dev.Host) +
                  "\",\"version\":\"" + String(VERSAO) +
                  "\",\"wifi\":" + String(WiFi.status() == WL_CONNECTED ? "true" : "false") +
                  ",\"mqtt\":" + String(ConexaoMQTT.connected() ? "true" : "false") + "}";
    server.send(200, "application/json", body);
  });

  MDNS.begin(dev.Host);

  atualizadorOTA.setup(&server);

  server.begin();

  MDNS.addService( "http", "tcp", 80);

  pinMode(PIN_BOTAO, INPUT_PULLUP);
  pinMode(PIN_RELE, OUTPUT);

  UltimoValor = digitalRead(PIN_BOTAO);
  releLigado = true;
  digitalWrite(PIN_RELE, ON);

  reconectaMQTT();
  RegistraSensor();

  solicitaLigamento();

  configTime(UTC_OFFSET, UTC_OFFSET_DST, NTP_SERVER);
  ultimaLeituraBotao =  millis();
}

// =====================================================
// LOOP
// =====================================================

void loop()
{
  server.handleClient();
  MDNS.update();

  if (ConexaoMQTT.connected())
  {
    ConexaoMQTT.loop();
    WatchdogMQTT();

    unsigned long agoraMQTT = millis();
    if (agoraMQTT - ultimaTelemetriaMQTT >= INTERVALO_TELEMETRIA_MQTT)
    {
      publicaTelemetriaMQTT();
      ultimaTelemetriaMQTT = agoraMQTT;
    }

    if (agoraMQTT - ultimoDiscoveryMQTT >= INTERVALO_DISCOVERY_MQTT)
    {
      RegistraSensor();
      ultimoDiscoveryMQTT = agoraMQTT;
    }
  }
  else
  {
    reconectaMQTT();
  }
  
  // ====================================
  // MQTT ON
  // ====================================

  if (mqttLiga)
  {
    mqttLiga = false;
    solicitaLigamento();
  }

  // ====================================
  // MQTT OFF
  // ====================================

  if (mqttDesliga)
  {
    mqttDesliga = false;
    solicitaDesligamento();
  }

  // ====================================
  // DESLIGAMENTO PROGRAMADO
  // ====================================

  if (desligamentoPendente && (long)(millis() - horarioDesligamento) >= 0)
  {
    desligamentoPendente = false;
    setRele(false);
  }

  lerBotao();
  //verificaBotaoReset();

  yield();
}
